cmake_minimum_required(VERSION 3.23)
project(geode-root-solver VERSION 0.2.0 LANGUAGES CXX)

# CMake searches the selected vcpkg triplet through its toolchain file.
find_path(GMP_INCLUDE_DIR NAMES gmp.h REQUIRED)
find_path(GMPXX_INCLUDE_DIR NAMES gmpxx.h REQUIRED)
find_path(MPFR_INCLUDE_DIR NAMES mpfr.h REQUIRED)
include(SelectLibraryConfigurations)
foreach(dependency IN ITEMS GMP GMPXX MPFR)
    string(TOLOWER "${dependency}" library_name)
    get_filename_component(dependency_prefix "${${dependency}_INCLUDE_DIR}" DIRECTORY)
    find_library(${dependency}_LIBRARY_RELEASE NAMES ${library_name}
        PATHS "${dependency_prefix}/lib" PATH_SUFFIXES "${CMAKE_LIBRARY_ARCHITECTURE}" NO_DEFAULT_PATH)
    find_library(${dependency}_LIBRARY_RELEASE NAMES ${library_name} REQUIRED)
    find_library(${dependency}_LIBRARY_DEBUG NAMES ${library_name}
        PATHS "${dependency_prefix}/debug/lib" NO_DEFAULT_PATH)
    select_library_configurations(${dependency})
endforeach()

add_library(geode_solver INTERFACE)
add_library(geode::solver ALIAS geode_solver)
target_compile_features(geode_solver INTERFACE cxx_std_17)
target_include_directories(geode_solver INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(geode_solver SYSTEM INTERFACE
    "${GMP_INCLUDE_DIR}" "${GMPXX_INCLUDE_DIR}" "${MPFR_INCLUDE_DIR}")
target_link_libraries(geode_solver INTERFACE ${MPFR_LIBRARIES} ${GMPXX_LIBRARIES} ${GMP_LIBRARIES})

option(GEODE_BUILD_EXAMPLES "Build the example program" ${PROJECT_IS_TOP_LEVEL})
if(GEODE_BUILD_EXAMPLES)
    add_executable(demo examples/main.cpp)
    target_link_libraries(demo PRIVATE geode::solver)
endif()

if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(geode_tests tests/solver_tests.cpp tests/header_link.cpp)
        target_link_libraries(geode_tests PRIVATE geode::solver)
        if(MSVC)
            target_compile_options(geode_tests PRIVATE /W4 /permissive-)
        else()
            target_compile_options(geode_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        endif()
        add_test(NAME geode_tests COMMAND geode_tests)
        set_tests_properties(geode_tests PROPERTIES TIMEOUT 120)
        if(GEODE_BUILD_EXAMPLES)
            add_test(NAME geode_demo COMMAND demo)
            set_tests_properties(geode_demo PROPERTIES TIMEOUT 30)
        endif()
    endif()
endif()
